        LIST    P=16F84A
        INCLUDE <P16F84A.INC>

        __CONFIG _XT_OSC & _WDT_OFF & _PWRTE_ON & _CP_OFF

;========================
; VARIABLES
;========================
        CBLOCK  0x0C
            CONT1
            CONT2
            CONT3
            VECES
        ENDC

        ORG     0x0000
        GOTO    INICIO

;========================
; INICIO
;========================
INICIO
        BSF     STATUS, RP0      ; Banco 1
        CLRF    TRISB            ; PORTB como salida
        BCF     STATUS, RP0      ; Banco 0

        CLRF    PORTB

MAIN
        MOVLW   D'3'             ; repetir 3 veces
        MOVWF   VECES

REPETIR
        MOVLW   B'01010101'      ; patrón 01010101
        MOVWF   PORTB
        CALL    RETARDO

        CLRF    PORTB            ; apagar todos
        CALL    RETARDO

        DECFSZ  VECES, F
        GOTO    REPETIR

        GOTO    MAIN

;========================
; RETARDO
;========================
RETARDO
        MOVLW   D'255'
        MOVWF   CONT1
R1
        MOVLW   D'255'
        MOVWF   CONT2
R2
        MOVLW   D'10'
        MOVWF   CONT3
R3
        DECFSZ  CONT3, F
        GOTO    R3
        DECFSZ  CONT2, F
        GOTO    R2
        DECFSZ  CONT1, F
        GOTO    R1
        RETURN

        END
