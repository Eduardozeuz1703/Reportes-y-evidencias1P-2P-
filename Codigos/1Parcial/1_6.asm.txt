        LIST    P=16F84A
        INCLUDE <P16F84A.INC>

        __CONFIG _XT_OSC & _WDT_OFF & _PWRTE_ON & _CP_OFF

;=======================
; VARIABLES
;=======================
        CBLOCK  0x0C
            CONT1
            CONT2
            VALOR
        ENDC

        ORG     0x0000
        GOTO    INICIO

;=======================
; INICIO
;=======================
INICIO
        BSF     STATUS, RP0      ; Banco 1
        CLRF    TRISB            ; PORTB salida
        BCF     STATUS, RP0      ; Banco 0

        MOVLW   B'00000001'      ; Inicia en RB0
        MOVWF   VALOR

;=======================
; LOOP PRINCIPAL
;=======================
MAIN
        MOVF    VALOR, W         ; W = VALOR
        MOVWF   PORTB            ; Muestra parte baja

        SWAPF   VALOR, W         ; Intercambia nibbles
        IORWF   PORTB, F         ; Une alto + bajo

        CALL    RETARDO

        ; Desplazar a la izquierda
        RLF     VALOR, F

        ; Si llega a 0 (overflow), reiniciar
        BTFSC   STATUS, Z
        MOVLW   B'00000001'
        BTFSC   STATUS, Z
        MOVWF   VALOR

        GOTO    MAIN

;=======================
; RETARDO
;=======================
RETARDO
        MOVLW   D'255'
        MOVWF   CONT1

R1
        MOVLW   D'255'
        MOVWF   CONT2

R2
        DECFSZ  CONT2, F
        GOTO    R2

        DECFSZ  CONT1, F
        GOTO    R1

        RETURN

        END
