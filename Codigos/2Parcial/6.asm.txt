LIST P=16F84A
INCLUDE <P16F84A.INC>

__CONFIG _XT_OSC & _WDT_OFF & _PWRTE_ON & _CP_OFF

CONTA1 EQU 0X0C
CONTA2 EQU 0X0D

ORG 0X00
GOTO INICIO

INICIO
    BSF STATUS,RP0
    CLRF TRISB
    BCF STATUS,RP0

    MOVLW 0X01
    MOVWF PORTB
    BCF STATUS,C

BUCLE
    CALL RETARDO
    RLF PORTB,F
    BTFSC STATUS,C
    BSF PORTB,0
    GOTO BUCLE

RETARDO
    MOVLW 0XFF
    MOVWF CONTA1
R1
    MOVLW 0XFF
    MOVWF CONTA2
R2
    DECFSZ CONTA2,F
    GOTO R2
    DECFSZ CONTA1,F
    GOTO R1
    RETURN

END
