        LIST    P=16F84A
        INCLUDE <P16F84A.INC>

        __CONFIG _XT_OSC & _WDT_OFF & _PWRTE_ON & _CP_OFF

        ORG     0x0000
        GOTO    INICIO

INICIO
        BSF     STATUS, RP0          ; Banco 1
        MOVLW   B'00000110'          ; RA1 y RA2 entradas
        MOVWF   TRISA
        CLRF    TRISB                ; PORTB salida
        BCF     STATUS, RP0          ; Banco 0

        CLRF    PORTB

LOOP
        BCF     PORTB,0              ; apaga por defecto

        BTFSC   PORTA,1              ; ¿RA1 = 1?
        GOTO    RA1_ACTIVO
        GOTO    RA1_CERO

RA1_ACTIVO
        BTFSS   PORTA,2              ; si RA2 = 0 -> encender
        GOTO    ENCENDER
        GOTO    APAGAR               ; si RA2 = 1 -> apagar

RA1_CERO
        BTFSC   PORTA,2              ; si RA2 = 1 -> encender
        GOTO    ENCENDER
        GOTO    APAGAR               ; si RA2 = 0 -> apagar

ENCENDER
        BSF     PORTB,0
        GOTO    LOOP

APAGAR
        BCF     PORTB,0
        GOTO    LOOP

        END
