LIST P=16F84A
INCLUDE <P16F84A.INC>

__CONFIG _XT_OSC & _WDT_OFF & _PWRTE_ON & _CP_OFF

AUX EQU 0X0C

ORG 0X00
GOTO INICIO

ORG 0X05
INICIO
    BSF STATUS,RP0
    MOVLW 0XFF
    MOVWF TRISA
    CLRF TRISB
    BCF STATUS,RP0

    CLRF PORTB

BUCLE
    MOVF PORTA,W
    MOVWF AUX
    BCF STATUS,C
    RLF AUX,F
    BSF AUX,0
    MOVF AUX,W
    MOVWF PORTB
    GOTO BUCLE

END
