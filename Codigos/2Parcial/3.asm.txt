LIST P=16F84A
INCLUDE <P16F84A.INC>

__CONFIG _XT_OSC & _WDT_OFF & _PWRTE_ON & _CP_OFF

SUMA1 EQU 0X0C

ORG 0X00
GOTO INICIO

ORG 0X05
INICIO
    BSF STATUS,RP0
    MOVLW 0XFF
    MOVWF TRISA
    CLRF TRISB
    BCF STATUS,RP0

    CLRF PORTA
    CLRF PORTB

BUCLE
    MOVF PORTA,W
    ADDWF PORTA,W
    MOVWF SUMA1

    MOVF PORTA,W
    ADDWF SUMA1,W
    MOVWF PORTB

    GOTO BUCLE

END
